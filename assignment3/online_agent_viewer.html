29def call_llm(messages: list, temperature: float = 0.3) -> str:
30 """Send a chat completion request to the LLM API."""
31 payload = json.dumps({
32 "model": MODEL,
33 "messages": messages,
34 "temperature": temperature,
35 "max_tokens": 2048,
36 }).encode("utf-8")
37
38 req = urllib.request.Request(API_URL, data=payload, method="POST")
39 req.add_header("Content-Type", "application/json")
40 req.add_header("Authorization", f"Bearer {API_KEY}")
41
42 try:
43 with urllib.request.urlopen(req, timeout=60) as resp:
44 body = json.loads(resp.read().decode())
45 return body["choices"][0]["message"]["content"]
46 except urllib.error.HTTPError as e:
47 return f"[API Error {e.code}] {e.read().decode()}"
48 except Exception as e:
49 return f"[Error] {str(e)}"

52def read_file(path: str, max_chars: int = 8000) -> str:
53 """Read a text file, truncating if too long."""
54 try:
55 with open(path, "r", encoding="utf-8", errors="replace") as f:
56 content = f.read()
57 if len(content) > max_chars:
58 content = content[:max_chars] + "\n\n[... truncated due to length ...]"
59 return content
60 except Exception as e:
61 return f"[Cannot read file] {e}"

64def analyze_file(file_path: str, user_question: str | None) -> str:
65 """Read a file and ask the LLM to analyze it."""
66 content = read_file(file_path)
67 question = user_question or "Please summarize this file and highlight the key points."
68
69 messages = [
70 {"role": "system", "content": "You are a helpful assistant. Analyze the provided document and answer the user's question accurately and concisely."},
71 {"role": "user", "content": f"Document content:\n\n{content}\n\n---\n\nQuestion: {question}"},
72 ]
73 return call_llm(messages)

76def web_search_agent(question: str) -> str:
77 """Ask the LLM a question that may require general knowledge (simulates web search via LLM)."""
78 messages = [
79 {"role": "system", "content": (
80 "You are a helpful research assistant with access to general knowledge up to 2024. "
81 "Answer the user's question thoroughly. If you are unsure about specific details, "
82 "state your level of confidence clearly. Structure your answer with bullet points "
83 "when listing multiple items."
84 )},
85 {"role": "user", "content": question},
86 ]
87 return call_llm(messages)
